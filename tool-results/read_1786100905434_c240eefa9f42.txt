   104→// ===== SCORE GAUGE COMPONENT =====
   105→function ScoreGauge({
   106→  score,
   107→  size = 120,
   108→  strokeWidth = 8,
   109→}: {
   110→  score: number;
   111→  size?: number;
   112→  strokeWidth?: number;
   113→}) {
   114→  const radius = (size - strokeWidth) / 2;
   115→  const circumference = 2 * Math.PI * radius;
   116→  const [offset, setOffset] = useState(circumference);
   117→
   118→  useEffect(() => {
   119→    const timer = setTimeout(() => {
   120→      setOffset(circumference * (1 - score / 100));
   121→    }, 100);
   122→    return () => clearTimeout(timer);
   123→  }, [score, circumference]);
   124→
   125→  return (
   126→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   127→      <svg width={size} height={size} className="-rotate-90">
   128→        {/* Background circle */}
   129→        <circle
   130→          cx={size / 2}
   131→          cy={size / 2}
   132→          r={radius}
   133→          fill="none"
   134→          stroke="#E5E7EB"
   135→          strokeWidth={strokeWidth}
   136→        />
   137→        {/* Score arc */}
   138→        <motion.circle
   139→          cx={size / 2}
   140→          cy={size / 2}
   141→          r={radius}
   142→          fill="none"
   143→          stroke="#D4AF37"
   144→          strokeWidth={strokeWidth}
   145→          strokeLinecap="round"
   146→          strokeDasharray={circumference}
   147→          initial={{ strokeDashoffset: circumference }}
   148→          animate={{ strokeDashoffset: offset }}
   149→          transition={{ duration: 1.2, ease: [0.22, 1, 0.36, 1], delay: 0.2 }}
   150→        />
   151→      </svg>
   152→      {/* Center score */}
   153→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   154→        <motion.span
   155→          className="font-[family-name:var(--font-jakarta)] font-bold text-navy"
   156→          style={{ fontSize: size * 0.28 }}
   157→          initial={{ opacity: 0, scale: 0.8 }}
   158→          animate={{ opacity: 1, scale: 1 }}
   159→          transition={{ duration: 0.5, delay: 0.6 }}
   160→        >
   161→          {score}
   162→        </motion.span>
   163→        <span
   164→          className="text-muted-foreground"
   165→          style={{ fontSize: size * 0.1 }}
   166→        >
   167→          /100
   168→        </span>
   169→      </div>
   170→    </div>
   171→  );
   172→}
